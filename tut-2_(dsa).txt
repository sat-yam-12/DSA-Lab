bool find(vector<int> & arr,int x){
    int n=arr.size();
    int low=0;
    int high=arr.size();
    while(low<=high){
         int mid=(low+high)/2;
        if(arr[mid]==x){
            return true;
        }
        if(arr[mid]>x){//  4>3
           high=mid-1;
        }
        else {
            low=mid+1;
        }
    }
    return false;
}

    vector<int> arr={1,2,3,4,5,6,7};

#include<iostream>

bool check(char c){
    c=tolower(c);
   return (c=='a' || c=='e' || c=='i' || c=='o' || c=='u');
}
int main(){
    string s1,result;
    cout<<"enter the string";
    getline(cin,s1);
    int len=s1.size();
    for(char i=0;i<len;i++){
        if(!check(s1[i])){
            result+=s1[i];
        }
    }

}

#include<iostream>
using namespace std;
void swap(char c,char d,string s){
    char temp=s[c];
    s[c]=s[d];
    s[d]=temp;
}
int main(){
    string s1;
    cout<<"enter the string \n";
    getline(cin,s1);
    int len=s1.size();
    for(int i=0;i<len;i++){
      for(int j=0;j<len-i-1;j++){
          if(s1[j]-'a'>s1[j+1]-'a'){
              swap(s1[j],s1[j+1]);
          }
      }
    }
    cout<<s1;
}

#include <stdio.h>

int main(){
    int n=5,i,j;

    printf("Diagonal:\n");
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            if(i==j) printf("A[%d]\t",i);
            else printf("0\t");
        }
        printf("\n");
    }

    printf("\nTri-diagonal:\n");
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            if(i==j+1) printf("A[%d]\t",i-1);
            else if(i==j) printf("A[%d]\t",(n-1)+i);
            else if(j==i+1) printf("A[%d]\t",(2*n-1)+i);
            else printf("0\t");
        }
        printf("\n");
    }

    printf("\nLower Triangular:\n");
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            if(j<=i) printf("A[%d]\t",i*(i+1)/2+j);
            else printf("0\t");
        }
        printf("\n");
    }

    printf("\nUpper Triangular:\n");
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            if(j>=i) printf("A[%d]\t",i*n - i*(i-1)/2 + (j-i));
            else printf("0\t");
        }
        printf("\n");
    }

    printf("\nSymmetric:\n");
    for(i=0;i<n;i++){
        for(j=0;j<n;j++){
            if(i>=j) printf("A[%d]\t",i*(i+1)/2+j);
            else printf("A[%d]\t",j*(j+1)/2+i);
        }
        printf("\n");
    }

    return 0;
}

// matrix multiplicstion
#include <bits/stdc++.h>
using namespace std;
struct Triple{int row,col,val;};
int main(){
    vector<Triple> A={{0,1,10},{0,3,12},{2,2,5},{3,1,15},{3,2,7}};
    vector<Triple> B={{0,2,8},{1,3,25},{2,0,9},{2,2,9},{3,0,20},{3,1,23}};
    vector<Triple> BT;
    for(auto &t:B) BT.push_back({t.col,t.row,t.val});
    int n=4; 
    vector<vector<int>> C(n,vector<int>(n,0));
    for(auto &a:A){
        for(auto &bt:BT){
            if(a.col==bt.col) C[a.row][bt.row]+=a.val*bt.val;
        }
    }
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++) cout<<setw(5)<<C[i][j]<<" ";
        cout<<endl;
    }
}
