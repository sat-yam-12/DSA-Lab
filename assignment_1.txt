#include <iostream>
using namespace std;
#define MAX 100
int arr[MAX];
int n = 0;
void create() {
    cout << "Enter number of elements: ";
    cin >> n;
    cout << "Enter elements: ";
    for (int i = 0; i < n; i++) {
        cin >> arr[i];
    }
}
void disp() {
    cout << "Array elements: ";
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}
void insertElement() {  
    int pos, el;
    cout << "Enter position and element: ";
    cin >> pos >> el;
    if (pos < 0 || pos > n) {
        cout << "Invalid position!" << endl;
        return;
    }
    for (int i = n; i > pos; i--) {
        arr[i] = arr[i - 1];
    }
    arr[pos] = el;
    n++;
}
void deleteElement() { 
    int pos;
    cout << "Enter position to delete: ";
    cin >> pos;
    if (pos < 0 || pos >= n) {
        cout << "Invalid position!" << endl;
        return;
    }
    for (int i = pos; i < n - 1; i++) {
        arr[i] = arr[i + 1];
    }
    n--;
}
void linearSearch() {  
    int key;
    cout << "Enter element to search: ";
    cin >> key;
    for (int i = 0; i < n; i++) {
        if (arr[i] == key) {
            cout << "Element found at index " << i << endl;
            return;
        }
    }
    cout << "Element not found!" << endl;
}

void menu(int choice) {
    switch (choice) {
        case 1: create(); break;
        case 2: disp(); break;
        case 3: insertElement(); break;
        case 4: deleteElement(); break;
        case 5: linearSearch(); break;
        case 6: cout << "Exiting...\n"; break;
        default: cout << "Invalid choice\n";
    }
}

int main() {
    int choice;
    cin>>choice;
    menu(choice);
}

#include <iostream>
using namespace std;
void duplicates(int arr[],int n){
    int mark[n];
   for (int i = 0; i < n; i++) {
        mark[i] = 1;
    }
    for(int i=0;i<n;i++){
        if(mark[i]==1){
            for(int j=i+1;j<n;j++){
                if(arr[i]==arr[j]){
                    mark[j]=0;
                }
            }
        }
    }
    for(int i=0;i<n;i++){
        if(mark[i]==1){
            cout<<arr[i];
        }
    }
}
int main(){
    int n=5;
    int arr[n]={1,2,1,2,1};
    duplicates(arr,n);
}

#include <iostream>
using namespace std;
int main() {
    int r1, c1, r2, c2;
    cout << "Enter rows and cols of first matrix: ";
    cin >> r1 >> c1;
    cout << "Enter rows and cols of second matrix: ";
    cin >> r2 >> c2;
    if (c1 != r2) {
        cout << "Matrix multiplication not possible!\n";
        return 0;
    }
    int A[r1][c1], B[r2][c2], C[r1][c2] = {0};
    cout << "Enter first matrix:\n";
    for (int i = 0; i < r1; i++)
        for (int j = 0; j < c1; j++)
            cin >> A[i][j];
    cout << "Enter second matrix:\n";
    for (int i = 0; i < r2; i++)
        for (int j = 0; j < c2; j++)
            cin >> B[i][j];
    // Multiplication
    for (int i = 0; i < r1; i++)
        for (int j = 0; j < c2; j++) {
            C[i][j] = 0;
            for (int k = 0; k < c1; k++)
                C[i][j] += A[i][k] * B[k][j];
        }
    cout << "Resultant Matrix:\n";
    for (int i = 0; i < r1; i++) {
        for (int j = 0; j < c2; j++)
            cout << C[i][j] << " ";
        cout << endl;
    }
}

#include <iostream>

int main() {
    int r, c;
    cout << "Enter rows and cols: ";
    cin >> r >> c;
    int A[r][c];
    cout << "Enter matrix:\n";
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++)
            cin >> A[i][j];

    cout << "Transpose:\n";
    for (int i = 0; i < c; i++) {
        for (int j = 0; j < r; j++)
            cout << A[j][i] << " ";
        cout << endl;
    }
}

#include <iostream>
using namespace std;

int main() {
    int r, c;
    cout << "Enter rows and cols: ";
    cin >> r >> c;
    int A[r][c];
    cout << "Enter matrix:\n";
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++)
            cin >> A[i][j];

    // Row sums
    for (int i = 0; i < r; i++) {
        int sum = 0;
        for (int j = 0; j < c; j++) sum += A[i][j];
        cout << "Sum of row " << i << " = " << sum << endl;
    }
    // Column sums
    for (int j = 0; j < c; j++) {
        int sum = 0;
        for (int i = 0; i < r; i++) sum += A[i][j];
        cout << "Sum of column " << j << " = " << sum << endl;
    }
}
